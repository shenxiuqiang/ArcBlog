## 5. 自检机检（**代码 + 命令 + 真实输出同处一篇**）

### 5.1 正本输出

```bash
$ python3 - <<'PY'
import io, os, re, glob, itertools
BASE = 'docs/07-团队与开发规范'
ALL = sorted(glob.glob(BASE + '/*.md'))
files = [p for p in ALL if not os.path.basename(p).startswith('_')]   # 过程件 `_提纲.md` 单列
TAGS = ['【已实现】','【规划中】','【平台边界】','【决策待定】','【未验证】']

# ⓪ 过程件：`_提纲.md` 只查行数上限（≤200），不参与"五类标签齐"
for p in [q for q in ALL if os.path.basename(q).startswith('_')]:
    n = len(io.open(p, encoding='utf-8').read().split('\n')) - 1
    print(f"{os.path.basename(p):34s} lines={n:4d} (过程件，上限 200)")

# ① 形态：行数 / 最长行（字符）/ 版本行
for p in files:
    lines = io.open(p, encoding='utf-8').read().split('\n')
    n = len(lines) - (1 if lines[-1] == '' else 0)
    mx = max(len(l) for l in lines)
    v = [len(l) for l in lines if l.startswith('| 版本 |')]
    print(f"{os.path.basename(p):34s} lines={n:4d} maxline={mx:4d} verline={v[0] if v else 'MISSING'}")

# ② 链接 0 MISS + 五类标签齐 + **归档引用**（R2：进 VERDICT）
miss, tags_bad, docs_links, docs_basis_bad = [], [], 0, []
for p in files:
    t = io.open(p, encoding='utf-8').read()
    for u in re.findall(r'\]\((\.\.?/[^)]+)\)', t):
        fp = os.path.normpath(os.path.join(BASE, u.split('#')[0]))
        if not os.path.exists(fp):
            miss.append((os.path.basename(p), u))
    if not all(tag in t for tag in TAGS):
        tags_bad.append(os.path.basename(p))
    n_docs = len(re.findall(r'\]\(\.\./\.\./_docs/', t))
    docs_links += n_docs
    # 引用 `_docs/**` 的文件，必须在同篇注明"归档依据/归档"（引用口径，不只是文件可达）
    if n_docs and '归档' not in t:
        docs_basis_bad.append(os.path.basename(p))
assert docs_links >= 1, "至少应有一处 `_docs/` 归档引用（否则 R2 的判据空转）"

# ③ 映射覆盖：14 / 13 / 13（由表行数生成，不手抄）
t = io.open(BASE + '/06-采纳与自检.md', encoding='utf-8').read()
sec = {}
for key, head, nxt in [('资产','## 1.','## 2.'), ('教训','## 2.','## 3.'), ('失效','## 3.','## 4.')]:
    blk = t.split(head)[1].split(nxt)[0]
    sec[key] = len([l for l in blk.split('\n') if re.match(r'^\| \d+ \|', l)])

# ④ 写范围不重叠（glob 交集；例外见 00 §2.2）
S = {'Lead':{'docs/README.md','docs/team/**'}, '主笔':{'docs/0X-*/**'},
     '审查者':{'docs/team/reviews/**'}, '调研员':{'docs/08-其他/调研/**'},
     'Dev':{'.aup/**','scripts/**','world/**','pages/**','.web/**','.route/**',
            'agents/**','config/**','seed/**','blocklet.yaml','package.json','dist/**'},
     'UI(职能)':{'docs/team/reviews/**'}, '人类':set(),
     '平台探针':{'.aup/**','scripts/**','world/**'}}
overlaps = []
ALLOWED = {('审查者', 'UI(职能)'): 'UI 并入审查者 = 同一域，非两人',
           ('Dev', '平台探针'): '平台探针与 Dev 同域，仅探针窗口内 + 持 .gate-lock'}
for a, b in itertools.combinations(S, 2):
    if S[a] & S[b] and (a, b) not in ALLOWED and (b, a) not in ALLOWED:
        overlaps.append((a, b, sorted(S[a] & S[b])))

print("links_missing", len(miss), miss[:3])
print("tags_incomplete", tags_bad)
print("_docs_links", docs_links, "| _docs_refs_without_basis", docs_basis_bad)
print("mapping_rows", sec, "expected", {'资产':14,'教训':13,'失效':13})
print("declared_exceptions", [f"{a}×{b}（{r}）" for (a, b), r in ALLOWED.items()])
print("scope_overlaps", overlaps)
print("VERDICT", "PASS" if (not miss and not tags_bad and not docs_basis_bad
      and sec == {'资产':14,'教训':13,'失效':13} and not overlaps) else "FAIL")
PY
```
