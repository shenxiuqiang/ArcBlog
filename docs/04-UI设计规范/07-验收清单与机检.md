## 2. 可机检条款 —— 代码 + 命令 + **真实输出**（本机实跑 2026-09-30；**本脚本共打印 17 项**，含 C03 的节点计数）

```bash
python3 - <<'ZZ'
import re,glob
files=[f for f in glob.glob('.aup/pages/*/*.aup')+glob.glob('.aup/pages/*.aup')]
t={f:open(f,encoding='utf-8').read() for f in files}
cnt=lambda p: sum(len(re.findall(p,v)) for v in t.values())
print('C02 pagination=auto =',cnt(r'pagination=auto'))
nodes=0; miss=[]
for f,v in t.items():
    ls=v.split('\n')
    for i,l in enumerate(ls):

            d=l.count('{')-l.count('}'); body=[l]; j=i+1
            while j<len(ls) and d>0: body.append(ls[j]); d+=ls[j].count('{')-ls[j].count('}'); j+=1
            nodes+=1
            if 'role=item' not in '\n'.join(body): miss.append(f'{f}:{i+1}')
print('C03 afs-list nodes =',nodes,'| without role=item =',len(miss))
print('C04 state={value:"${ =',cnt(r'state=\{value: *"\\$\{'))

print('C07 forbidden primitives =',sum(len(re.findall(r'(?<![a-z-])(badge|card|col)\s*\{',v)) for v in t.values()))
print('S01 自造空值 =',cnt(r'暂无数据|尚无数据|No data'))
print('S02 emptyText=: =',cnt(r'emptyText=:'))
print('S03 visible=!${state. =',cnt(r'visible="!\$\{state\.'))
print('S04 visible=!$state. =',cnt(r'visible="!\$state\.'))
print('S06 loading= =',cnt(r'loading='))
pb=[f for f,v in t.items() if 'propBind=' in v]; doc=[f for f in pb if re.search(r'render-timing|two-stage',t[f])]
print('S05 propBind pages =',len(pb),'documented =',len(doc))
w=[f for f,v in t.items() if '/.actions/write' in v or '/.actions/delete' in v]
print('S07 write pages =',len(w),'missing onErrorToast =',len([f for f in w if 'onErrorToast' not in t[f]]))
print('T02 runtime key build =',cnt(r'\$t\([a-zA-Z0-9_-]+\.\$\{'))
print('T06 english literal =',cnt(r'label="(SEO Title|OG )'))
bad=[]
for f,v in t.items():
    pid=re.search(r'page\s+([a-z0-9-]+)',v); ns=set(re.findall(r'\$t\(([a-z0-9-]+)\.',v))-{'wrapper'}
    if pid and ns-{pid.group(1)}: bad.append(f.split('/')[-1])
print('T03 namespace-leak pages =',len(bad),bad)
print('V01 hardcoded colors =',len([1 for v in t.values() for l in v.split('\n') if re.search(r'"#[0-9a-fA-F]{3,6}"',l)]))
ZZ
```
